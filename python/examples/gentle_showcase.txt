# RobotTwin gentle showcase — a flowing demo that stays easy on the drivetrain.
# Run:  python python/run_script.py --port COM5 python/examples/gentle_showcase.txt
# Preview (no hardware):
#   python python/run_script.py --dry-run --start 0,90,-90 python/examples/gentle_showcase.txt

PING
ENABLE

# Datum: place the arm at base 0deg, shoulder 90deg (upper arm vertical),
# elbow -90deg before running.
SETHOME 0 90 -90
PAYLOAD 0
TELEM 2

# 1) a small bow hello
MOVEJ 0 80 -105
SLEEP 1
MOVEJ 0 90 -90
SLEEP 1

# 2) rise into a relaxed, folded "ready" pose (arm up, elbow tucked in)
MOVEJ 0 105 -105
SLEEP 1

# 3) slow pan left, full sweep right, back to center -- the dramatic part
MOVEJ -80 105 -105
SLEEP 1
MOVEJ 80 105 -105
SLEEP 1
MOVEJ 0 105 -105
SLEEP 1

# 4) reach forward and draw back (elbow only half-opens, shoulder stays
#    near vertical so the load stays over the base)
MOVEJ 0 100 -55
SLEEP 1
MOVEJ 0 100 -110
SLEEP 1

# 5) two diagonal flourishes across the workspace
MOVEJ 60 95 -75
SLEEP 1
MOVEJ -60 95 -105
SLEEP 1

# 6) settle home and rest the motors
MOVEJ 0 90 -90
SLEEP 1
TELEM 0
DISABLE
